```python
    def forecast_table_rows(self) -> list[dict[str, str]]:

            weekly_hist = self._weekly_actual_series_for("hdan")
            if not weekly_hist:

            rows: list[dict[str, str]] = []

                week_date = last_date + pd.DateOffset(weeks=step)
                # "month" key kept even for weekly rows (not renamed to "week") so

                # via row["month"] -- only the header LABEL and column SET branch.

                for series_key in WEEKLY_FORECAST_SERIES_LABELS:
                    series = results.get(series_key, [])
                    entry = series[step - 1] if step - 1 < len(series) else None
                    for scenario in ("base", "bull", "bear"):
                        value = entry[scenario] if entry is not None else None
                        row[f"{series_key}_{scenario}"] = (
                            f"{value:,.2f}" if value is not None else ""
                        )
                rows.append(row)
```

```python
                    ),

    )
```

```python
        (DashboardState.forecast_warning != "") & (DashboardState.granularity == "monthly"),
        rx.hstack(
            rx.icon("triangle-alert", size=16, color="amber"),
            rx.text(DashboardState.forecast_warning, size=RADIX_SIZE_BODY, color="amber"),
            spacing="2",
            align="center",
            margin_bottom="0.75rem",
        ),
        rx.fragment(),
    )
    return rx.box(
        warning_banner,
        rx.cond(
            DashboardState.forecast_table_rows.length() > 0,
            table,
            rx.text(
                rx.cond(
                    DashboardState.granularity == "weekly",

                    DashboardState.forecast_error,
                ),

        ),
        background=DashboardState.surface,

        border_radius=CARD_RADIUS,
        padding=CARD_PADDING,
        overflow_x="auto",
        width="100%",
        min_width="0",
    )
```
